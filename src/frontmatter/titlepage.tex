% src/frontmatter/titlepage.tex
% Cover front: cover-figure.pdf ocupa TODA la primera página letterpage (612x792 pt).
% Overlay: título centrado abajo, cita + versión en las esquinas inferiores.
% Esto resuelve la queja original: "el cover era un sol pequeño a la derecha; quiero una portada real tipo CARTA".

\begin{titlepage}
	% Evita anclas duplicadas de hyperref en esta primera página
	\hypersetup{pageanchor=false}
	\newgeometry{margin=0pt}

	% Paleta del cover front (mismo lenguaje visual del título original)
	\definecolor{CanvasDark}{HTML}{0A0F1A} % fondo muy oscuro
	\definecolor{SunCore}{HTML}{DC2626}    % rojo cálido (núcleo)
	\definecolor{SunRing}{HTML}{FCA5A5}    % halo claro rojizo

	\begin{tikzpicture}[remember picture, overlay]

		% --- LAYER 1: dark canvas (fondo de respaldo si la imagen no carga) ---
		\fill[CanvasDark] (current page.south west) rectangle (current page.north east);

		% --- LAYER 2: cover-figure.pdf llena toda la página ---
		% Escala exacto a paperwidth × paperheight (letterpage = 612 × 792 pt).
		% Fallback chain: cover-figure.pdf → cover.pdf → cover-figure.png → cover.png.
		\IfFileExists{assets/cover-figure.pdf}{%
			\node[anchor=south west, inner sep=0pt, outer sep=0pt]
				at (current page.south west)
				{\includegraphics[width=\paperwidth,height=\paperheight,keepaspectratio=false]{assets/cover-figure.pdf}};
		}{%
			\IfFileExists{assets/cover.pdf}{%
				\node[anchor=south west, inner sep=0pt, outer sep=0pt]
					at (current page.south west)
					{\includegraphics[width=\paperwidth,height=\paperheight,keepaspectratio=false]{assets/cover.pdf}};
			}{%
				\IfFileExists{assets/cover-figure.png}{%
					\node[anchor=center, inner sep=0pt, outer sep=0pt]
						at (current page.center)
						{\includegraphics[width=\paperwidth]{assets/cover-figure.png}};
				}{%
					\IfFileExists{assets/cover.png}{%
						\node[anchor=center, inner sep=0pt, outer sep=0pt]
							at (current page.center)
							{\includegraphics[width=\paperwidth]{assets/cover.png}};
					}{%
						% Placeholder sobrio si no se encuentra la imagen
						\draw[SunRing, line width=1pt] ($(current page.center)+(-4cm,-2cm)$) rectangle ($(current page.center)+(4cm,2cm)$);
						\node[anchor=center, text=SunRing] at (current page.center)
							{\small\sffamily Coloca aquí cover-figure.pdf (en src/assets/)};
					}%
				}%
			}%
		}

		% --- LAYER 3: velo sutil oscuro SOLO en la franja inferior ---
		% Hace el texto superpuesto legible sin alterar la imagen principal.
		\shade[top color=CanvasDark!0, bottom color=CanvasDark!75,
		       middle color=CanvasDark!20]
			($(current page.south west)+(0,0)$)
			rectangle
			($(current page.north east)+(0,-0.42\paperheight)$);

		% --- LAYER 4: filete dorado separador título / pie ---
		\draw[SunCore, line width=1pt, opacity=0.85]
			($(current page.west)+(1.6cm,-0.43\paperheight+0.6cm)$) --
			($(current page.east)+(-1.6cm,-0.43\paperheight+0.6cm)$);

		% --- LAYER 5: TÍTULO + SUBTÍTULO sobre el velo inferior ---
		\node[anchor=center, text=white, align=center]
			at ($(current page.south)+(0,0.34\paperheight)$) {%
				\sffamily\bfseries
				\fontsize{34}{40}\selectfont Geometría de la Información\\[6pt]
				\fontsize{14}{17}\selectfont\mdseries
				Un mapa para comparar modelos%
			};

		% --- LAYER 6: PIE — versión / fecha (izquierda) + cita (derecha) ---
		\node[anchor=west, xshift=1.6cm, yshift=1.6cm, text=SunRing]
			at (current page.south west)
			{\small \textit{Edición} \BookVersion\ (\BookDate)};

		\node[anchor=east, xshift=-1.6cm, yshift=1.6cm, text=SunRing, align=right]
			at (current page.south east)
			{\footnotesize\itshape ``El sol es nuevo cada día.''\\\itshape --- Heráclito};

	\end{tikzpicture}

	\vfill
	\restoregeometry
	\hypersetup{pageanchor=true}
\end{titlepage}
