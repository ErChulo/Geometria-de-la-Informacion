% src/main.tex — archivo principal (pdfLaTeX, twoside para impresión)
\documentclass[12pt,twoside]{book}

% Márgenes asimétricos (para encuadernación)
\usepackage{geometry}
\geometry{letterpaper, inner=1.75in, outer=1in, top=1in, bottom=2in, heightrounded}

% Versión del libro (define \BookVersion y \BookDate)
\input{version}

% Estilo general (tipos, colores, títulos, encabezados)
\usepackage{style-book}

% Carpeta de imágenes para todo el libro
\usepackage{graphicx}
\graphicspath{{assets/}}

\begin{document}\frontmatter
	\input{frontmatter/titlepage}

	% -----------------------------------------------------------------
	% Blank verso page between cover and TOC.
	% In twoside book class, the first page after the cover (page ii)
	% can inherit a quirky \textheight from the titlepage's
	% \newgeometry/\restoregeometry dance, which clips the TOC at the
	% bottom edge. Inserting a blank verso pushes the TOC onto page iii
	% (recto) where the geometry is clean. \thispagestyle{empty}
	% suppresses the page number that would otherwise print on the
	% otherwise-empty verso.
	% -----------------------------------------------------------------
	\thispagestyle{empty}\null\newpage
	\cleardoublepage

	\setcounter{tocdepth}{1}
	\begingroup
		\raggedbottom              % don't force TOC to stretch to bottom
		\emergencystretch=4em      % give LaTeX room to reflow
		\tableofcontents
		\cleardoublepage
	\endgroup
	\markboth{}{}    % limpia marcas de TOC
	\pagestyle{fancy}
	
	% Prólogo en frontmatter (números romanos, sin capítulo numerado)
	\include{Prologo}
	\cleardoublepage
	
	\mainmatter
	
	% Asegura que el siguiente capítulo sea el 0 y muestre número grande
	\setcounter{chapter}{0}
	\include{capitulo0}
	\include{capitulo1}
	\include{capitulo2}
	\include{capitulo3}
	\include{capitulo4}
	\include{capitulo5}
	\include{capitulo6}
	\include{capitulo7}
	\include{capitulo8}	\backmatter
	% Bibliografía / apéndices (si aplica)
	\include{respuestas}
	\include{apendice_mathematica}
	\include{bibliografia}

	% Back cover (eclipse solar con coqui-negro.jpg) — última página del libro
	\cleardoublepage
	\input{frontmatter/backcover}

\end{document}
